// Example: tiny debounce utility
function debounce(fn, ms=100){
  let t; return (...args)=>{ clearTimeout(t); t=setTimeout(()=>fn(...args), ms)}
}

// Usage:
const onResize = debounce(()=>console.log('resized'), 200);
window.addEventListener('resize', onResize);
